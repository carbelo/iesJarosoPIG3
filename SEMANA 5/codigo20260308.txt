// callbacks wifi
void callBackWifiEvent(void* handler_arg, esp_event_base_t base, int32_t id, void* event_data)
{
    if (id == WIFI_EVENT_STA_START)
    {
        ESP_LOGI(TAG, "WiFi STA iniciado, conectando...");
        esp_wifi_connect();
    }
    else if (id == WIFI_EVENT_STA_CONNECTED)
    {
        ESP_LOGI(TAG, "WiFi conectado!");
    }
    else if (id == WIFI_EVENT_STA_DISCONNECTED)
    {
        ESP_LOGI(TAG, "WiFi desconectado, reintentando...");
        esp_wifi_connect();
    }
}


static bool humedadTaskCreated = false;

void callBackIpEvent(void* handler_arg, esp_event_base_t base, int32_t id, void* event_data)
{
    if(id == IP_EVENT_STA_GOT_IP)
    {
        ESP_LOGI(TAG, "IP obtenida, conectando a broker MQTT...");
        const esp_mqtt_client_config_t mqtt_cfg = {
        .broker.address.uri = "mqtt://10.0.7.248",
        };
        h_mqttClient = esp_mqtt_client_init(&mqtt_cfg);
        esp_mqtt_client_register_event(h_mqttClient,ESP_EVENT_ANY_ID, callBackMqttEvent, NULL);
        esp_mqtt_client_start(h_mqttClient);

        if (!humedadTaskCreated) {
            ESP_LOGI(TAG, "Creando tareas de humedad");
            xTaskCreate( tareaSenHumedad1, "tareasSenHumedad1", SIZETAREAS, NULL, 1, &h_tareaSenHumedad1);
            xTaskCreate( tareaSenHumedad2, "tareasSenHumedad2", SIZETAREAS, NULL, 1, &h_tareaSenHumedad2);
            humedadTaskCreated = true;
        }
    }
}
void callBackMqttEvent(void* handler_arg, esp_event_base_t base, int32_t id, void* event_data)
{
    esp_mqtt_event_t * datoRecibido;
    datoRecibido = (esp_mqtt_event_t*)event_data;

    if(id == MQTT_EVENT_CONNECTED)
    {
        ESP_LOGI(TAG, "MQTT conectado al broker");
        esp_mqtt_client_subscribe_single(h_mqttClient, "4/1/0", 0);
        ESP_LOGI(TAG, "Suscrito a topic 4/1/0");
        esp_mqtt_client_subscribe_single(h_mqttClient, "4/2/0", 0);
        ESP_LOGI(TAG, "Suscrito a topic 4/2/0");
        esp_mqtt_client_subscribe_single(h_mqttClient, "4/3/0", 0);
        ESP_LOGI(TAG, "Suscrito a topic 4/3/0");
    }
    else if(id == MQTT_EVENT_SUBSCRIBED)
    {
        ESP_LOGI(TAG, "Suscripcion a topics completada");
    }
    else if(id == MQTT_EVENT_DISCONNECTED)
    {
        ESP_LOGI(TAG, "MQTT desconectado del broker");
    }
    else if (id == MQTT_EVENT_DATA)
    {
        ESP_LOGI(TAG, "Dato MQTT recibido - Topic: %.*s", datoRecibido->topic_len, datoRecibido->topic);
        if (datoRecibido->data_len > 0)
        {
            int valor = (datoRecibido->data[0] == '1') ? 1 : 0;

            if (datoRecibido->topic_len == 5 && strncmp(datoRecibido->topic, "4/1/0", 5) == 0)
            {
                ESP_LOGI(TAG, "EV1: %d", valor);
                xQueueSend(h_colaEV1, &valor, 0);
            }
            else if (datoRecibido->topic_len == 5 && strncmp(datoRecibido->topic, "4/2/0", 5) == 0)
            {
                ESP_LOGI(TAG, "EV2: %d", valor);
                xQueueSend(h_colaEV2, &valor, 0);
            }
            else if (datoRecibido->topic_len == 5 && strncmp(datoRecibido->topic, "4/3/0", 5) == 0)
            {
                ESP_LOGI(TAG, "Bomba: %d", valor);
                xQueueSend(h_colaBomba, &valor, 0);
            }
        }
    }
}

    ESP_LOGI(TAG, "Tarea Bomba iniciada");

void tareaSenCaudalimetro( void * pvParameters )
{
    ESP_LOGI(TAG, "Tarea Caudalimetro iniciada");
    //Configuración sensor caudal1 gpio 36
    gpio_config_t configSenCaudal1 = {
        .pin_bit_mask = MaskPinCaudalimetro,
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_ENABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    gpio_config(&configSenCaudal1);
    
    pcnt_unit_config_t unitConfig = {
        .high_limit = CAUDAL1CONTADORMAX,
        .low_limit = CAUDAL1CONTADORMIN,
    };
    pcnt_new_unit(&unitConfig, &h_unitCaudal1);
    
    pcnt_chan_config_t configChannel = {
        .edge_gpio_num = PinCaudalimetro,
        .level_gpio_num = -1,
        .flags = {
            .virt_level_io_level = 1
        }
    };
    pcnt_new_channel(h_unitCaudal1, &configChannel, &h_channelCaudal1);
    pcnt_channel_set_edge_action(h_channelCaudal1, PCNT_CHANNEL_EDGE_ACTION_HOLD, PCNT_CHANNEL_EDGE_ACTION_INCREASE);
    pcnt_channel_set_level_action(h_channelCaudal1, PCNT_CHANNEL_LEVEL_ACTION_KEEP, PCNT_CHANNEL_LEVEL_ACTION_KEEP);

    pcnt_unit_enable(h_unitCaudal1);
    pcnt_unit_start(h_unitCaudal1);

    esp_timer_create_args_t configTimer10sCaudal1 = {
        .callback = &cb_timer10sCaudal1,
        .name = "timer10sCaudal1"
    };
    esp_timer_create(&configTimer10sCaudal1, &h_timer10sCaudal1);
    esp_timer_start_periodic(h_timer10sCaudal1, 10000000);
    ESP_LOGI(TAG, "Timer de caudalimetro iniciado");

    while(1) 
    { 
        vTaskDelay(portMAX_DELAY);
    }
}

void cb_timer10sCaudal1(void * args) 
{
    int pulses = 0;
    packetSend_t datoEnviar = {
        .grupo = 4,
        .id = 12,
        .funcion = 0,
        .readWrite = 0,
        .dato = 0
    };
    
    pcnt_unit_get_count(h_unitCaudal1, &pulses);
    pcnt_unit_clear_count(h_unitCaudal1);
    datoEnviar.dato = pulses;
    
    if (h_colaEnvioMqtt != NULL) {
        xQueueSend(h_colaEnvioMqtt, &datoEnviar, 0);
    }
    ESP_LOGI(TAG, "Caudal - Pulsos en 10s: %d", pulses);
}

void tareaSenHumedad1( void * pvParameters )
{
    ESP_LOGI(TAG, "Tarea Humedad 1 iniciada");
    packetSend_t datoEnviar = {
        // crear paquete de datos a enviar por mqtt
        .grupo = 4,
        .id = 10,
        .funcion = 0,
        .readWrite = 0,
        .dato = 0
    };

    gpio_config_t configSenHumedad1 = {
        .pin_bit_mask = MaskPinSensorHumedad1,
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE
    };
    gpio_config(&configSenHumedad1);

    adc_oneshot_chan_cfg_t cfg = {
        .bitwidth = ADC_BITWIDTH_10,
        .atten = ADC_ATTEN_DB_12,
    };
    esp_err_t err = adc_oneshot_config_channel(h_cadUnit1, ADC_CHANNEL_6, &cfg);
    if (err != ESP_OK) {
        return;
    }

    while(1)
    {
        int val = 0;
        adc_oneshot_read(h_cadUnit1, ADC_CHANNEL_6, &val);
        
        int pct;
        if (val >= 740) pct = 0;
        else if (val <= 350) pct = 100;
        else pct = (740 - val) * 100 / 390;
        
        datoEnviar.dato = pct;
        ESP_LOGI(TAG, "Humedad 1 - CAD: %d, Porcentaje: %d%%", val, pct);
        xQueueSend(h_colaEnvioMqtt, &datoEnviar, portMAX_DELAY);
        vTaskDelay(pdMS_TO_TICKS(100000));
    }
}

void tareaEnvioMqtt( void * pvParameters )
{
    ESP_LOGI(TAG, "Tarea Envio MQTT iniciada");
    packetSend_t msg;
    char top[10];
    char dat[10];
    while(1)
    {
        xQueueReceive(h_colaEnvioMqtt, &msg, portMAX_DELAY);
        // crear topics con grupo
        sprintf(top, "%d/%d/%d", msg.grupo, msg.id, msg.funcion);
        // si es sensor de humedad, agregar '%' al valor (quitar para mandar el crudo)
        if (msg.id == 10 || msg.id == 11) {
            snprintf(dat, sizeof(dat), "%d%%", msg.dato);
        } else {
            snprintf(dat, sizeof(dat), "%d", msg.dato);
        }
        if (h_mqttClient != NULL) {
            esp_mqtt_client_publish(h_mqttClient, top, dat, strlen(dat), 0, 0);
            ESP_LOGI(TAG, "MQTT publicado - Topic: %s, Dato: %s", top, dat);
        }
        xSemaphoreGive(h_semaforoEnvioMqtt);
    }
}
