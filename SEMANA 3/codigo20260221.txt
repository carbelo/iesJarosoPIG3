void callBackWifiEvent(void* handler_arg, esp_event_base_t base, int32_t id, void* event_data)

        ESP_LOGI(TAG, "Función wifi se ha iniciado");

    }
   
}

// Conectar ak broker MQTT al obtener IP
void callBackIpEvent(void* handler_arg, esp_event_base_t base, int32_t id, void* event_data)
{
   
    if(id == IP_EVENT_STA_GOT_IP)
    {
        ESP_LOGI(TAG, "ip obtenida");
        const esp_mqtt_client_config_t mqtt_cfg = {
        .broker.address.uri = "mqtt://10.0.7.248",
        };
        h_mqttClient = esp_mqtt_client_init(&mqtt_cfg);
        esp_mqtt_client_register_event(h_mqttClient,ESP_EVENT_ANY_ID, callBackMqttEvent, NULL);
        esp_mqtt_client_start(h_mqttClient);


    }
}
void callBackMqttEvent(void* handler_arg, esp_event_base_t base, int32_t id, void* event_data)
{
    esp_mqtt_event_t * datoRecibido;
    datoRecibido = (esp_mqtt_event_t*)event_data;

    if(id == MQTT_EVENT_CONNECTED)
    {
        ESP_LOGI(TAG, "MQTT: conectado al broker");
        esp_mqtt_client_subscribe_single(h_mqttClient, "1/0", 0);
        esp_mqtt_client_subscribe_single(h_mqttClient, "2/0", 0);
    }
    else if( id == MQTT_EVENT_SUBSCRIBED)
    {
        ESP_LOGI(TAG, "MQTT: unido a topic");
    }
    else if (id == MQTT_EVENT_DATA)
    {
        ESP_LOGI(TAG, "MQTT DATA - Topic: %.*s, Payload: %.*s, Len: %d",
                 datoRecibido->topic_len, datoRecibido->topic,
                 datoRecibido->data_len, datoRecibido->data,
                 datoRecibido->data_len);
        
        if (datoRecibido->data_len > 0)
        {
            int valor = (datoRecibido->data[0] == '1') ? 1 : 0;

            // Suscribirse a los topics
            if (datoRecibido->topic_len == 3 && strncmp(datoRecibido->topic, "1/0", 3) == 0)
            {
                if (xQueueSend(h_colaEV1, &valor, 0) == pdPASS)
                {
                    ESP_LOGI(TAG, "Comando EV1 enviado a cola: %d", valor);
                }
                else
                {
                    ESP_LOGI(TAG, "Error: Cola EV1 llena");
                }
            }
            else if (datoRecibido->topic_len == 3 && strncmp(datoRecibido->topic, "2/0", 3) == 0)
            {
                if (xQueueSend(h_colaEV2, &valor, 0) == pdPASS)
                {
                    ESP_LOGI(TAG, "Comando EV2 enviado a cola: %d", valor);
                }
                else
                {
                    ESP_LOGI(TAG, "Error: Cola EV2 llena");
                }
            }
            else
            {
                ESP_LOGI(TAG, "Topic no reconocido: %.*s", datoRecibido->topic_len, datoRecibido->topic);
            }
        }
        else
        {
            ESP_LOGI(TAG, "Payload vacío");
        }
    }
}

// Funciones de los sensores y actuadores
void tareaActBomba( void * pvParameters )
{
    int dato;

    while(1)
    {
        xQueueReceive( h_colaBomba, &dato, portMAX_DELAY);
        ESP_LOGI(TAG, "DATO TAREA BOMBA %d", dato);
    }
}

void tareaActEV1( void * pvParameters )
{
    int dato;
    
    while(1)
    {
        // Esperar por comando en la cola
        if (xQueueReceive(h_colaEV1, &dato, portMAX_DELAY) == pdPASS)
        {
            if (dato == 1)
            {
                gpio_set_level(PinActElectroValvula1, 1);
                ESP_LOGI(TAG, "TareaActEV1: Relé ACTIVADO (GPIO %d nivel 1)", PinActElectroValvula1);
            }
            else if (dato == 0)
            {
                gpio_set_level(PinActElectroValvula1, 0);
                ESP_LOGI(TAG, "TareaActEV1: Relé DESACTIVADO (GPIO %d nivel 0)", PinActElectroValvula1);
            }
        }
    }
}

void tareaEnvioMqtt( void * pvParameters )
{

    packetSend_t datoRecibido;
    while(1)
    {

    }
